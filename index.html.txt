    const CONSOLES = [

      { id: 'nds', name: 'Nintendo DS', free: false, icon: 'fa-tablet-screen-button' },
      { id: 'n64', name: 'Nintendo 64', free: false, icon: 'fa-dice-d6' },
      { id: 'psx', name: 'PlayStation 1', free: false, icon: 'fa-compact-disc' }
    ];

    let currentUser = {
      email: null,
      isLoggedIn: false,
      isVIP: false
    };

    let activeSystem = 'citra_3ds';
    let crtEnabled = false;
    let lastGeneratedCode = null;
    let isDrawing = false;

    // --- CONSOLE GRID BUILDER ---
    function renderConsoleGrid() {
      const grid = document.getElementById('console-grid');
      grid.innerHTML = '';

      CONSOLES.forEach(c => {
        const isLocked = !c.free && !currentUser.isVIP;
        const isSelected = activeSystem === c.id;

        const btn = document.createElement('button');
        btn.className = `p-3 rounded-xl border text-left transition flex flex-col justify-between h-20 ${
          isSelected 
            ? 'bg-cyan-950/70 border-cyan-500 shadow-md shadow-cyan-500/20' 
            : 'bg-slate-950 border-slate-800 hover:border-slate-700'
        }`;

        btn.onclick = () => selectConsole(c.id);

        btn.innerHTML = `
          <div class="flex items-center justify-between w-full">
            <i class="fa-solid ${c.icon} ${isSelected ? 'text-cyan-400' : 'text-slate-400'} text-xs"></i>
            ${isLocked ? '<span class="text-[9px] bg-amber-500/10 text-amber-400 border border-amber-500/30 px-1.5 py-0.5 rounded font-bold"><i class="fa-solid fa-lock text-[8px]"></i> VIP</span>' : '<span class="text-[9px] bg-emerald-500/10 text-emerald-400 border border-emerald-500/30 px-1.5 py-0.5 rounded font-bold">FREE</span>'}
          </div>
          <div>
            <div class="text-xs font-bold text-slate-200 truncate">${c.name}</div>
          </div>
        `;
        grid.appendChild(btn);
      });
    }

    // --- SELECT CONSOLE LOGIC ---
    function selectConsole(id) {
      const consoleData = CONSOLES.find(c => c.id === id);
      
      // If locked and user not VIP
      if (!consoleData.free && !currentUser.isVIP) {
        showToast(`${consoleData.name} requires VIP access!`, true);
        requestVIPModal();
        return;
      }

      activeSystem = id;
      document.getElementById('active-system-name').innerText = consoleData.name;
      
      // Switch view mode: 3DS dual-screen vs retro single screen
      const view3DS = document.getElementById('view-3ds');
      const viewRetro = document.getElementById('view-retro');

      if (id === 'citra_3ds') {
        view3DS.classList.remove('hidden');
        viewRetro.classList.add('hidden');
        document.getElementById('system-avatar').innerText = '3DS';
        document.getElementById('active-system-status').innerText = 'Dual-Screen Stereoscopic Engine Active';
      } else {
        view3DS.classList.add('hidden');
        viewRetro.classList.remove('hidden');
        document.getElementById('system-avatar').innerText = id.toUpperCase().slice(0, 3);
        document.getElementById('active-system-status').innerText = 'Universal Core Ready';
        renderRetroDemo();
      }

      renderConsoleGrid();
      playTone(520, 'sine', 0.1);
    }

    // --- AUTHENTICATION ENGINE ---
    function quickLogin(email) {
      currentUser.email = email;
      currentUser.isLoggedIn = true;
      
      // Auto-VIP Developer Condition
      if (email.toLowerCase() === 'cooooooool67@gmail.com') {
        currentUser.isVIP = true;
        showToast("Master VIP Developer account activated! All consoles unlocked.");
      } else {
        currentUser.isVIP = false;
        showToast(`Signed in as ${email}`);
      }

      closeModal('auth-modal');
      updateUserUI();
      renderConsoleGrid();
      playTone(660, 'triangle', 0.15);
    }

    function submitEmailLogin() {
      const email = document.getElementById('auth-email-input').value.trim();
      if (!email || !email.includes('@')) {
        showToast("Please enter a valid email address.", true);
        return;
      }
      quickLogin(email);
    }

      currentUser = { email: null, isLoggedIn: false, isVIP: false };

          vipPromo.classList.remove('hidden');
        }

        navArea.innerHTML = `
          <div class="flex items-center gap-2">

    // --- CITRA 3DS DUAL SCREEN RENDERING & STYLUS ---
    let topCtx, btmCtx, retroCtx;
    let starfield = [];

    function initCanvases() {
      const top = document.getElementById('topScreenCanvas');
      const btm = document.getElementById('bottomScreenCanvas');
      const retro = document.getElementById('retroCanvas');

      topCtx = top.getContext('2d');
      btmCtx = btm.getContext('2d');
      retroCtx = retro.getContext('2d');

      // Generate stars for top 3D screen animation
      for (let i = 0; i < 60; i++) {
        starfield.push({
          x: Math.random() * 400,
          y: Math.random() * 240,
          speed: 0.5 + Math.random() * 2,
          size: Math.random() * 2
        });
      }

      // Draw bottom touch screen greeting
      resetBottomCanvas();

      // Stylus touch event listeners on bottom screen
      btm.addEventListener('mousedown', (e) => startStroke(e, btm));
      btm.addEventListener('mousemove', (e) => drawStroke(e, btm));
      window.addEventListener('mouseup', () => isDrawing = false);

      btm.addEventListener('touchstart', (e) => {
        const touch = e.touches[0];
        const mouseEvent = new MouseEvent("mousedown", { clientX: touch.clientX, clientY: touch.clientY });
        btm.dispatchEvent(mouseEvent);
      });
      btm.addEventListener('touchmove', (e) => {
        const touch = e.touches[0];
        const mouseEvent = new MouseEvent("mousemove", { clientX: touch.clientX, clientY: touch.clientY });
        btm.dispatchEvent(mouseEvent);
        e.preventDefault();
      });
      btm.addEventListener('touchend', () => isDrawing = false);

      requestAnimationFrame(renderLoop);
    }

    function resetBottomCanvas() {
      btmCtx.fillStyle = '#0f172a';
      btmCtx.fillRect(0, 0, 320, 240);

      // Grid pattern
      btmCtx.strokeStyle = '#1e293b';
      btmCtx.lineWidth = 1;
      for (let x = 0; x < 320; x += 20) {
        btmCtx.beginPath(); btmCtx.moveTo(x, 0); btmCtx.lineTo(x, 240); btmCtx.stroke();
      }
      for (let y = 0; y < 240; y += 20) {
        btmCtx.beginPath(); btmCtx.moveTo(0, y); btmCtx.lineTo(320, y); btmCtx.stroke();
      }

      btmCtx.fillStyle = '#38bdf8';
      btmCtx.font = '12px Inter';
      btmCtx.fillText("Nintendo 3DS Stylus Touch Digitizer", 20, 30);
      btmCtx.fillStyle = '#94a3b8';
      btmCtx.font = '10px Inter';
      btmCtx.fillText("Drag mouse or finger here to draw / calibrate", 20, 50);
    }

    function startStroke(e, canvas) {
      isDrawing = true;
      drawStroke(e, canvas);
      playTone(400, 'triangle', 0.04);
    }

    function drawStroke(e, canvas) {
      if (!isDrawing) return;
      const rect = canvas.getBoundingClientRect();
      const scaleX = canvas.width / rect.width;
      const scaleY = canvas.height / rect.height;
      const x = (e.clientX - rect.left) * scaleX;
      const y = (e.clientY - rect.top) * scaleY;

      btmCtx.fillStyle = '#f59e0b';
      btmCtx.beginPath();
      btmCtx.arc(x, y, 3, 0, Math.PI * 2);
      btmCtx.fill();
    }

    // --- AUDIO SYNTH (Web Audio API) ---
    let audioCtx;
    function initAudioEngine() {
      window.addEventListener('click', () => {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }, { once: true });
    }

    function playTone(freq, type = 'sine', duration = 0.1) {
      if (!audioCtx) return;
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch (e) {}
    }
